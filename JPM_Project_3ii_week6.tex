% ============================== 10. STEP 6 PAYOFF ENGINE ==============================
\begin{frame}{Step 6 --- Payoff engine prices both election choices}{SEC fields provide legal terms; WRDS fields provide prices.}
  \vspace{0.35cm}
  \begin{tabularx}{\linewidth}{@{}>{\bfseries\color{navy}}p{3.4cm} X@{}}

    Menu values & $V_{\text{cash}} = C = \$152.27;\qquad V_{\text{stock}} = r \times P_A = \$140.08$\\[7pt]
    Favored election & $\mathrm{argmax}(V_{\text{cash}},\, V_{\text{stock}}) = \text{cash}$\\[7pt]

  \end{tabularx}

% ============================== 11. STEP 7 PRORATION ==============================
\begin{frame}{Step 7 --- Proration model converts demand into EV and alpha}{The smoke test uses simple v1 assumptions so the full data loop can be verified first.}
  \vspace{0.25cm}
  \begin{tabularx}{\linewidth}{@{}>{\bfseries}p{3.2cm} X@{}}
    \rowcolor{navy}\hdr{Calculation} & \hdr{Smoke-test output}\\ \midrule
    Passive block & $p = 17.7\%$\\
    Predicted cash demand & $D_{\text{cash}} = (1-p)\times 85\% + p\times 10\% = 71.7\%$\\
    Cash fill rate & $f_{\text{cash}} = \min(1,\ 15.0\%/71.7\%) = 20.9\%$\\
    Expected value & $EV = 20.9\%\times\$152.27 + 79.1\%\times\$140.08 = \$142.63$\\
    Net alpha & $\$142.63 - \$138.32 - 10\text{bp cost} = \$4.17$ per share\\
  \end{tabularx}
  \vspace{0.4cm}
  \begin{center}\small\color{slate}
    V1 constants: borrow cost = 0; borrow availability = 1; short interest = 0.
  \end{center}
\end{frame}
