\begin{frame}{This Week's Message}
\small
\textbf{The pipeline now runs end to end on an actual election deal.}

\vspace{0.6em}
\begin{itemize}
  \item We can identify whether a merger is truly an election/proration deal.
  \item We can retrieve the relevant public SEC files instead of manually searching deal documents.
  \item We can send those files to Claude and receive the canonical fields needed for the project.
  \item We can join WRDS market data and ETF/index-fund ownership data onto the same event.
  \item The resulting event row is sufficient to compute alpha and construct the election-arbitrage strategy.
\end{itemize}

\vspace{0.7em}
\textbf{Smoke test used in this update:} Andeavor / Marathon Petroleum, announced April 30, 2018.
\end{frame}

\begin{frame}{End-to-End Pipeline}
\small
\begin{enumerate}
  \item \textbf{Deal screen:} identify candidate M\&A events and reject non-election deals.
  \item \textbf{SEC retrieval:} download S-4, 424B3, 8-K, and later 10-Q/10-K filings around the transaction timeline.
  \item \textbf{Claude extraction:} submit the relevant filing package and map text evidence into canonical deal fields.
  \item \textbf{WRDS join:} add announcement-date market variables and ETF/index-fund ownership.
  \item \textbf{Strategy table:} output coverage, predicted value, alpha, hedge ratio, and trade signal.
\end{enumerate}

\vspace{0.7em}
\scriptsize
Main scripts: \texttt{download\_ma\_edgar\_files.py}, \texttt{download\_ownership\_etf\_data.py}, \texttt{download\_wrds\_market\_data.py}, \texttt{build\_election\_strategy\_model.py}.
\end{frame}

\begin{frame}{Formula Bridge: Fields To Payoffs}
\small
\textbf{The extracted fields map directly into the two election payoffs.}

\vspace{0.4em}
\begin{align*}
C &= \text{cash consideration per target share} = \usd{152.27} \\
r &= \text{stock exchange ratio} = 1.87 \\
P_A &= \text{acquirer price at signal date} = \usd{74.91} \\
P_T &= \text{target price at signal date} = \usd{138.32}
\end{align*}

\vspace{0.2em}
\begin{align*}
V_{cash} &= C = \usd{152.27} \\
V_{stock} &= r \times P_A = 1.87 \times \usd{74.91} = \usd{140.08} \\
\text{favored election} &= \arg\max(V_{cash}, V_{stock}) = \text{cash}
\end{align*}

\vspace{0.3em}
\textbf{Pipeline point.} SEC supplies $C$ and $r$; WRDS supplies $P_A$ and $P_T$; the joined row can price both choices.
\end{frame}

\begin{frame}{Formula Bridge: Proration To EV}
\small
\textbf{The strategy table then converts election demand and caps into expected consideration.}

\vspace{0.3em}
\begin{align*}
K_{cash} &= \text{cash cap fraction} = 15.0\% \\
D_{cash} &= \text{predicted cash-election demand} = 71.7\% \\
f_{cash} &= \min\left(1,\frac{K_{cash}}{D_{cash}}\right)
          = \min\left(1,\frac{15.0\%}{71.7\%}\right)
          = 20.9\%
\end{align*}

\vspace{0.2em}
\begin{align*}
EV
&= f_{cash} \times V_{cash} + (1-f_{cash}) \times V_{stock} \\
&= 20.9\% \times \usd{152.27}
   + 79.1\% \times \usd{140.08} \\
&= \usd{142.63}
\end{align*}

\vspace{0.2em}
\textbf{Readout.} If a cash election is only partially filled, the unfilled piece receives stock consideration.
\end{frame}

\begin{frame}{Alpha Calculation Can Be Produced}
\small
\textbf{Alpha is calculated directly from expected consideration value versus the current target price.}

\vspace{0.4em}
\begin{align*}
\alpha_{gross}
&= EV - P_T \\
&= \usd{142.63} - \usd{138.32} \\
&= \usd{4.31}\ \text{per ANDV share}
\end{align*}

\vspace{0.2em}
\begin{align*}
\text{Trading cost} &= 10\text{ bps} \times P_T
                    = 0.001 \times \usd{138.32}
                    = \usd{0.14} \\
\text{Borrow cost} &= \usd{0.00}\ \text{in the current smoke test}
\end{align*}

\vspace{0.2em}
\[
\alpha_{net}
= \usd{4.31} - \usd{0.14} - \usd{0.00}
= \usd{4.17}\ \text{per ANDV share}
\]
\end{frame}

\begin{frame}{Arbitrage Strategy Row Can Be Produced}
\small
\textbf{The same row also gives the hedge ratio and implementable paper trade.}

\vspace{0.4em}
\begin{align*}
\text{Target leg} &= +1.000\ \text{ANDV share} \\
\text{Election instruction} &= \text{elect cash} \\
\text{Expected stock receipt} &= (1-f_{cash}) \times r \\
&= (1-20.9\%) \times 1.87 \\
&= 1.479\ \text{MPC shares per ANDV share}
\end{align*}

\vspace{0.2em}
\[
\text{Hedged position}
= +1.000\ \text{ANDV}
-1.479\ \text{MPC}
+ \text{cash election right}
\]

\vspace{0.3em}
\textbf{Strategy output.} Favored election = cash; hedge ratio = 1.479 MPC short per ANDV long; signal = \texttt{paper\_trade}.
\end{frame}
